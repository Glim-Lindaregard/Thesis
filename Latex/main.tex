\newcommand{\thr}[5]{% corner, dx, dy, state(on/off), \TL
        \pgfmathsetmacro{\dx}{#2}
        \pgfmathsetmacro{\dy}{#3}
        \path (#1) coordinate (C);
        \path let \p1 = (C) in coordinate (T) at (\x1 - #5*\dx, \y1 - \TL*\dy);
        \ifthenelse{\equal{#4}{on}}{%
            \draw[thrOn] (C) -- (T);
        }{%
            \draw[thrOff] (C) -- (T);
        }%
    }
